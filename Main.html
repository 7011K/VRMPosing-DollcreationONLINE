async function loadAndInitAddons({ threeRenderer, mountPoint }) { // ...（略）アドオンJSのimportと登録... window.addonHandle = () => { const targetButton = document.querySelector('.MuiButtonBase-root.MuiIconButton-root.MuiIconButton-sizeMedium.css-1egpgfe'); const root = document.getElementById("root"); if (!targetButton) { console.warn("対象ボタン (.main-action-button) が見つかりません"); return; } if (!addonVisible) { if (targetButton.dataset.addonIconsInjected) return; targetButton.dataset.addonIconsInjected = "true"; loadedAddons.forEach((addon, index) => { const icon = document.createElement("button"); icon.className = "addon-selector-button"; icon.title = `Addon ${index + 1}`; icon.innerText = "★"; icon.dataset.addonIndex = index; icon.addEventListener("click", (e) => { e.stopPropagation(); const fn = registeredAddonFns[index]; if (typeof fn === "function") { fn({ threeRenderer: null, mountPoint: root, addonBaseUrl: addon.baseUrl }); } }); targetButton.parentNode.insertBefore(icon, targetButton); }); document.addEventListener("click", outsideClickHandler); addonVisible = true; } else { hideAddonButtons(); } function outsideClickHandler(e) { const isAddonButton = e.target.classList.contains("addon-selector-button"); const isHandleButton = e.target === targetButton; if (!isAddonButton && !isHandleButton) { hideAddonButtons(); } } function hideAddonButtons() { const icons = document.querySelectorAll(".addon-selector-button"); icons.forEach(btn => btn.remove()); targetButton.dataset.addonIconsInjected = ""; document.removeEventListener("click", outsideClickHandler); addonVisible = false; } }; // ここに追加！ window._loadedAddons = loadedAddons; window._registeredAddonFns = registeredAddonFns; }
