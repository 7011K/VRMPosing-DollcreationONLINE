window.addonHandle = () => { console.log("[addonHandle] 実行開始"); // 全ての該当ボタンを取得 const buttons = document.querySelectorAll('.MuiButtonBase-root.MuiIconButton-root.MuiIconButton-sizeMedium.css-1egpgfe'); console.log("[addonHandle] 該当ボタン数:", buttons.length); let targetButton = null; // aria-label="アドオン"のボタンだけをターゲットに buttons.forEach(btn => { console.log("[addonHandle] チェック中 aria-label:", btn.getAttribute("aria-label")); if (btn.getAttribute('aria-label') === 'アドオン') { targetButton = btn; } }); if (!targetButton) { console.warn('[addonHandle] aria-label="アドオン" のボタンが見つかりません'); return; } console.log('[addonHandle] ターゲットボタン取得:', targetButton); // 一度消す hideAbsoluteAddonButtons(); // 座標取得 const rect = targetButton.getBoundingClientRect(); console.log('[addonHandle] ターゲットボタン座標:', rect); const BUTTON_SIZE = 32; // ボタン高さ(px) const BUTTON_MARGIN = 6; // ボタン間隔(px) // 配置座標計算（左側に並べる例） let left = rect.left - (loadedAddons.length * (BUTTON_SIZE + BUTTON_MARGIN)); let top = rect.top; loadedAddons.forEach((addon, index) => { console.log('[addonHandle] ★ボタン生成開始 index:', index, 'addon:', addon); const icon = document.createElement("button"); icon.className = "addon-selector-button-absolute"; icon.title = `Addon ${index + 1}`; icon.innerText = "★"; icon.dataset.addonIndex = index; // 位置計算・設定 const iconLeft = left + index * (BUTTON_SIZE + BUTTON_MARGIN); icon.style.left = iconLeft + "px"; icon.style.top = (top) + "px"; icon.style.height = BUTTON_SIZE + "px"; icon.style.width = BUTTON_SIZE + "px"; console.log(`[addonHandle] ★ボタン配置 left:${iconLeft} top:${top}`); icon.addEventListener("click", (e) => { e.stopPropagation(); console.log(`[addonHandle] ★ボタン${index}クリック`); const fn = registeredAddonFns[index]; if (typeof fn === "function") { console.log(`[addonHandle] ★ボタン${index}のアドオン関数呼び出し`); fn({ threeRenderer: null, mountPoint: document.getElementById("root"), addonBaseUrl: addon.baseUrl }); } else { console.warn(`[addonHandle] ★ボタン${index}の関数が未定義`); } }); icon.setAttribute("data-addon-absolute", "true"); document.body.appendChild(icon); console.log('[addonHandle] ★ボタン追加完了 index:', index); }); // イベント: レイアウト変化で消す（resize/scroll/クリック） window.addEventListener("resize", hideAbsoluteAddonButtons, { once: true }); window.addEventListener("scroll", hideAbsoluteAddonButtons, { once: true }); document.addEventListener("click", hideAbsoluteAddonButtons, { once: true }); console.log("[addonHandle] ★ボタン群表示完了"); }; function hideAbsoluteAddonButtons() { console.log("[hideAbsoluteAddonButtons] ★ボタン群消去開始"); document.querySelectorAll('.addon-selector-button-absolute').forEach(btn => btn.remove()); window.removeEventListener("resize", hideAbsoluteAddonButtons, { once: true }); window.removeEventListener("scroll", hideAbsoluteAddonButtons, { once: true }); document.removeEventListener("click", hideAbsoluteAddonButtons, { once: true }); console.log("[hideAbsoluteAddonButtons] ★ボタン群消去完了"); }
